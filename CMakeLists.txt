cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(TarkovLootPrices VERSION 0.3.0 LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "Windows 10/11 only")
endif()
include(FetchContent)
FetchContent_Declare(nlohmann_json URL
  https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
add_executable(TarkovLootPrices WIN32 src/main.cpp)
target_compile_features(TarkovLootPrices PRIVATE cxx_std_20)
target_compile_definitions(TarkovLootPrices PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(TarkovLootPrices PRIVATE user32 gdi32 winhttp shell32 nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(TarkovLootPrices PRIVATE /W4 /permissive- /utf-8 /EHsc)
endif()
include(CTest)
if(BUILD_TESTING)
  foreach(test price_math catalog icon_scan)
    add_executable(${test}_tests tests/${test}_tests.cpp)
    target_compile_features(${test}_tests PRIVATE cxx_std_20)
    target_include_directories(${test}_tests PRIVATE src)
    target_link_libraries(${test}_tests PRIVATE nlohmann_json::nlohmann_json)
    add_test(NAME ${test} COMMAND ${test}_tests)
  endforeach()
endif()
